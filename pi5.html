# Inisialisasi Jendela Utama
window = tk.Tk()

window.geometry("350x200")

# Variabel Status
led_state = False

def toggle_led():
    global led_state
    led_state = not led_state
    if led_state:
        label_status.config(text="Status LED: MENYALA", fg="green")
        btn_control.config(text="MATIKAN LED")
    else:
        label_status.config(text="Status LED: MATI", fg="red")
        btn_control.config(text="NYALAKAN LED")

# Label Status
label_status = tk.Label(window, text="Status LED: MATI", fg="red", font=("Arial", 10))
label_status.pack(pady=5)

# Tombol Eksekusi
btn_control = tk.Button(window, text="NYALAKAN LED", command=toggle_led, bg="#2563eb", fg="white", font=("Arial", 10, "bold"))
btn_control.pack(pady=15)

# Jalankan Loop Utama GUI
window.mainloop()

        nano gui_gpio_integration.py

        import tkinter as tk
import RPi.GPIO as GPIO
import threading
import time

# Konfigurasi Hardware
LED_PIN = 18
GPIO.setmode(GPIO.BCM)
GPIO.setup(LED_PIN, GPIO.OUT)
GPIO.output(LED_PIN, GPIO.LOW)

# GUI Setup
root = tk.Tk()
root.title("Monitoring & Control System")
root.geometry("400x250")

is_running = True

# Komponen GUI
lbl_title = tk.Label(root, text="Integrasi Sensor & Aktuator", font=("Arial", 14, "bold"))
lbl_title.pack(pady=10)

lbl_status = tk.Label(root, text="LED Status: OFF", font=("Arial", 11), fg="red")
lbl_status.pack(pady=5)

def toggle_hardware():
    current_state = GPIO.input(LED_PIN)
    new_state = not current_state
    GPIO.output(LED_PIN, new_state)
    
    if new_state:
        lbl_status.config(text="LED Status: ON", fg="green")
        btn_toggle.config(text="Turn OFF")
    else:
        lbl_status.config(text="LED Status: OFF", fg="red")
        btn_toggle.config(text="Turn ON")

btn_toggle = tk.Button(root, text="Turn ON", command=toggle_hardware, width=15, height=2)
btn_toggle.pack(pady=15)

def on_closing():
    global is_running
    is_running = False
    GPIO.cleanup()
    root.destroy()

root.protocol("WM_DELETE_WINDOW", on_closing)
root.mainloop()
