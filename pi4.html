        nano mqtt_pub.py

        import paho.mqtt.client as mqtt
import time
import json
import random

BROKER = "broker.hivemq.com"
PORT = 1883
TOPIC = "untirta/elektro/iot/sensor"

client = mqtt.Client()
client.connect(BROKER, PORT, 60)

print(f"Mulai mempublikasikan data MQTT ke Broker: {BROKER}...")

try:
    while True:
        payload = {
            "device": "RaspberryPi_Node1",
            "suhu": round(random.uniform(26.0, 33.0), 1),
            "status": "NORMAL"
        }
        json_data = json.dumps(payload)
        client.publish(TOPIC, json_data)
        print(f"[PUBLISH] Topic: {TOPIC} | Data: {json_data}")
        time.sleep(2)
except KeyboardInterrupt:
    client.disconnect()
    print("\nPengiriman data MQTT dihentikan.")
