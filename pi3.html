        import smbus
import time

# Alamat I2C umum (misal: LCD I2C = 0x27, BME280 = 0x76, PCF8574 = 0x27)
I2C_ADDR = 0x27
bus = smbus.SMBus(1)

print(f"Mencoba berkomunikasi dengan perangkat I2C di alamat 0x{I2C_ADDR:02X}...")

try:
    # Membaca 1 byte data dari register 0x00
    data = bus.read_byte_data(I2C_ADDR, 0x00)
    print(f"Berhasil membaca data I2C: {hex(data)}")
except Exception as e:
    print("Gagal membaca bus I2C. Periksa kembali pengkabelan dan alamat I2C perangkat Anda.")
    print("Detail Error:", e)

        nano spi_adc.py

        import spidev
import time

# Inisialisasi SPI (Bus 0, Device CE0)
spi = spidev.SpiDev()
spi.open(0, 0)
spi.max_speed_hz = 1350000

def read_adc(channel):
    if channel < 0 or channel > 7:
        return -1
    # Protokol MCP3008: Start bit + Single-ended bit + Channel select
    adc = spi.xfer2([1, (8 + channel) << 4, 0])
    data = ((adc[1] & 3) << 8) + adc[2]
    return data

try:
    print("Membaca data ADC MCP3008 via SPI (Ctrl+C untuk keluar)...")
    while True:
        val = read_adc(0)
        voltage = (val * 3.3) / 1023.0
        print(f"Channel 0 Raw: {val} | Tegangan: {voltage:.2f}V")
        time.sleep(0.5)
except KeyboardInterrupt:
    spi.close()
    print("\nKomunikasi SPI dihentikan.")
